<script>
    // Notun password set kora holo
    const ADMIN_PASS = "v3dk41sw";

    function checkPassword() {
        const input = document.getElementById('passInput').value;
        if (input === ADMIN_PASS) {
            document.getElementById('loginBox').style.display = 'none';
            document.getElementById('adminBox').style.display = 'block';
            loadManageList();
        } else {
            alert("ভুল পাসওয়ার্ড!");
        }
    }

    function getVideos() {
        const saved = localStorage.getItem('portal_videos');
        return saved ? JSON.parse(saved) : [];
    }

    function loadManageList() {
        const videos = getVideos();
        const listContainer = document.getElementById('videoManageList');
        listContainer.innerHTML = '';

        videos.forEach(v => {
            const item = document.createElement('div');
            item.className = 'video-list-item';
            item.innerHTML = `
                <div>
                    <strong style="color:#fff;">${v.title}</strong><br>
                    <small style="color:#b8b2c8;">👁️ ${v.views} | ❤️ ${v.likes}</small>
                </div>
                <button class="del-btn" onclick="deleteVideo(${v.id})">রিমুভ</button>
            `;
            listContainer.appendChild(item);
        });
    }

    function readFileAsDataURL(file) {
        return new Promise((resolve) => {
            if (!file) resolve("");
            const reader = new FileReader();
            reader.onload = () => resolve(reader.result);
            reader.readAsDataURL(file);
        });
    }

    document.getElementById('logoForm').addEventListener('submit', async function(e) {
        e.preventDefault();
        const logoFile = document.getElementById('logoFile').files[0];
        const logoUrl = document.getElementById('logoUrl').value;

        let finalLogoSrc = logoUrl;
        if (logoFile) {
            finalLogoSrc = await readFileAsDataURL(logoFile);
        }

        if (finalLogoSrc) {
            localStorage.setItem('portal_logo', finalLogoSrc);
            alert("লোগো সফলভাবে আপডেট করা হয়েছে!");
        } else {
            alert("দয়া করে ফাইল সিলেক্ট করুন অথবা সঠিক লিংক দিন।");
        }
    });

    document.getElementById('uploadForm').addEventListener('submit', async function(e) {
        e.preventDefault();

        const videoFile = document.getElementById('vFile').files[0];
        const videoUrl = document.getElementById('vUrl').value;
        const thumbFile = document.getElementById('tFile').files[0];
        const thumbUrl = document.getElementById('tUrl').value;

        let finalVideoSrc = videoUrl;
        if (videoFile) finalVideoSrc = await readFileAsDataURL(videoFile);

        let finalThumbSrc = thumbUrl;
        if (thumbFile) finalThumbSrc = await readFileAsDataURL(thumbFile);

        const videos = getVideos();
        const newVideo = {
            id: Date.now(),
            title: document.getElementById('vTitle').value,
            thumbnail: finalThumbSrc || "uploads/thumbnails/thumb_1784344858_8726.jpg",
            videoSrc: finalVideoSrc,
            views: document.getElementById('vViews').value,
            likes: document.getElementById('vLikes').value,
            comments: document.getElementById('vComments').value
        };

        videos.unshift(newVideo);
        localStorage.setItem('portal_videos', JSON.stringify(videos));
        document.getElementById('uploadForm').reset();
        loadManageList();
        alert("ভিডিও সফলভাবে আপলোড করা হয়েছে!");
    });

    function deleteVideo(id) {
        if (confirm("ভিডিওটি ডিলিট করতে চান?")) {
            let videos = getVideos();
            videos = videos.filter(v => v.id !== id);
            localStorage.setItem('portal_videos', JSON.stringify(videos));
            loadManageList();
        }
    }
</script>
