<script>
    const ADSTERRA_DIRECT_LINK = "https://www.profitableratecpmnetwork.com/xyze22qb1?key=472659c5ea753bdbd2c2cb94751c9ac6";
    let selectedVideoData = null;

    function loadSiteLogo() {
        const savedLogo = localStorage.getItem('portal_logo');
        if (savedLogo) {
            document.getElementById('siteLogoImg').src = savedLogo;
            document.getElementById('faviconLogo').href = savedLogo;
        }
    }

    function getStoredVideos() {
        const saved = localStorage.getItem('portal_videos');
        return saved ? JSON.parse(saved) : [
            {
                id: 1,
                title: "রোমান্টিক ভাইরাল এক্সক্লুসিভ ১৮+ ক্লিপ 🥵",
                thumbnail: "uploads/thumbnails/thumb_1784344858_8726.jpg",
                videoSrc: "https://www.w3schools.com/html/mov_bbb.mp4",
                views: "483.5K", likes: "58K", comments: "4.1K"
            }
        ];
    }

    function renderVideoGrid() {
        loadSiteLogo();
        const videos = getStoredVideos();
        const grid = document.getElementById('videoGrid');
        grid.innerHTML = '';
        videos.forEach(video => {
            const card = document.createElement('div');
            card.className = 'video-card';
            card.onclick = () => openUnlockModal(video.id);
            card.innerHTML = `
                <div class="thumbnail-box">
                    <img src="${video.thumbnail || 'uploads/thumbnails/thumb_1784344858_8726.jpg'}" alt="Thumb">
                    <div class="play-btn-overlay"><i class="fas fa-play" style="color:#fff;"></i></div>
                </div>
                <div class="video-info">
                    <div class="video-title">${video.title}</div>
                    <div class="video-stats">
                        <span>👁️ ${video.views}</span><span>❤️ ${video.likes}</span>
                    </div>
                </div>
            `;
            grid.appendChild(card);
        });
    }

    function openUnlockModal(videoId) {
        const videos = getStoredVideos();
        selectedVideoData = videos.find(v => v.id === videoId);
        if (!selectedVideoData) return;

        sessionStorage.setItem('active_video_id', videoId);
        document.getElementById('modalVideoTitle').innerText = selectedVideoData.title;

        if (sessionStorage.getItem('ad_clicked_' + videoId) === 'true') {
            showPlayerDirectly();
        } else {
            document.getElementById('unlockSection').style.display = 'block';
            document.getElementById('playerBox').style.display = 'none';
        }

        document.getElementById('unlockModal').style.display = 'flex';
    }

    function showPlayerDirectly() {
        document.getElementById('unlockSection').style.display = 'none';
        document.getElementById('playerBox').style.display = 'block';
        const player = document.getElementById('mainVideoPlayer');
        player.src = selectedVideoData.videoSrc || "https://www.w3schools.com/html/mov_bbb.mp4";
    }

    function closeModal() {
        document.getElementById('unlockModal').style.display = 'none';
        document.getElementById('mainVideoPlayer').pause();
    }

    function handleSingleAdClick() {
        window.open(ADSTERRA_DIRECT_LINK, '_blank');

        if (selectedVideoData) {
            sessionStorage.setItem('ad_clicked_' + selectedVideoData.id, 'true');
        }

        showPlayerDirectly();
    }

    window.onload = function() {
        renderVideoGrid();
        const activeVideoId = sessionStorage.getItem('active_video_id');
        const adClicked = sessionStorage.getItem('ad_clicked_' + activeVideoId);
        if (activeVideoId && adClicked === 'true') {
            openUnlockModal(Number(activeVideoId));
        }
    };
</script>
